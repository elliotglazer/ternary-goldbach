/-! Source transport for the same complete eight-by-129 literal trace.
The imported computed arithmetic modules are part of the measured closure. -/

namespace RawStateComputed
open RawHead RawStateData RawStateSources

theorem certificate_1 : traceCertificate initial_1 step_1 trace_1 = true := by
  rw [← trace_checked_1]
  exact traceCertificate_cachedTrace _ _ _

theorem actual_trace_1 : rawCachedTrace 0 1 128 = trace_1 := by
  change cachedTrace (rawStateInitial 1) (rawStepNumerator 0 1) 128 = _
  have hi : rawStateInitial 1 = initial_1 := initial_checked_1
  rw [hi, step_checked_1]
  exact trace_checked_1

theorem actual_error_1 (j : ℕ) (hj : j < 129) :
    ‖expIntegerValue (2 ^ 120) (trace_1[j]'(by rw [trace_length_1]; exact hj)) -
      integerPower (1 : ℝ) ((j : ℝ) * 5 / 64)‖ < 1 / (2 : ℝ) ^ 96 := by
  exact traceCertificate_actual_error
    (show initial_1 = rawStateInitial 1 from initial_checked_1.symm)
    step_checked_1.symm trace_1 certificate_1
    (by decide) (by decide) (by decide) (by decide) j
    (by rw [trace_length_1]; exact hj) (by omega)
