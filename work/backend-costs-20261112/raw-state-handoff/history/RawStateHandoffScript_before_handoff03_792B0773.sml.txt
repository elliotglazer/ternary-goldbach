(* Exact integer handoff only. No HOL logarithm, power or source enclosure. *)
Theory RawStateHandoff
Ancestors BlockRound integer list pair option
Libs intLib cv_transLib

val _ = intLib.prefer_int ();

Definition raw_mul_def:
  raw_mul (ur,ui) (vr,vi) =
    (round_scaled 5316911983139663491615228241121378304
       (ur*vr-ui*vi),
     round_scaled 5316911983139663491615228241121378304
       (ur*vi+ui*vr))
End

Definition raw_trace_def:
  (raw_trace 0 w z = [z]) /\
  (raw_trace (SUC n) w z = z::raw_trace n w (raw_mul z w))
End

Definition raw_produce_def:
  (raw_produce n [] = []) /\
  (raw_produce n ((z,w)::xs) = raw_trace n w z::raw_produce n xs)
End

Definition raw_head_def:
  raw_head (c:int) (ur,ui) (vr,vi) =
    (round_scaled 1393796574908163946345982392040522594123776
       (c*(ur*vr+ui*vi)),
     round_scaled 1393796574908163946345982392040522594123776
       (c*(ur*vi-ui*vr)))
End

Definition raw_heads_def:
  (raw_heads c [] vs = []) /\
  (raw_heads c (u::us) [] = []) /\
  (raw_heads c (u::us) (v::vs) =
     raw_head c u v::raw_heads c us vs)
End

(* Total indexing: exact dimensions are checked independently before use. *)
Definition raw_get_def:
  (raw_get n [] = []) /\
  (raw_get 0 (x::xs) = x) /\
  (raw_get (SUC n) (x::xs) = raw_get n xs)
End

Definition raw_coord_ok_def:
  raw_coord_ok (r:int,i:int) <=>
    -2658455991569831745807614120560689152 < r /\
    r < 2658455991569831745807614120560689152 /\
    -2658455991569831745807614120560689152 < i /\
    i < 2658455991569831745807614120560689152
End

Definition raw_shape_def:
  raw_shape traces <=> LENGTH traces = 8 /\
    EVERY (\row. LENGTH row = 129 /\ EVERY raw_coord_ok row) traces
End

Definition raw_consume_def:
  raw_consume c traces =
    if raw_shape traces then
      let den = raw_get 4 traces in
      SOME [raw_heads c den (raw_get 0 traces);
            raw_heads c den (raw_get 1 traces);
            raw_heads c den (raw_get 2 traces);
            raw_heads c den (raw_get 3 traces)]
    else NONE
End

Definition raw_pipeline_def:
  raw_pipeline (bases:num list) c n sources =
    if bases = [1;2;3;41;61;10004;262144;265800] /\
       c = 61 /\ n = 128 then
      raw_consume c (raw_produce n sources)
    else NONE
End

Theorem raw_trace_length:
  !n w z. LENGTH (raw_trace n w z) = SUC n
Proof
  Induct >> simp [raw_trace_def]
QED

Theorem raw_produce_length:
  !sources. LENGTH (raw_produce n sources) = LENGTH sources
Proof
  Induct >> simp [raw_produce_def] >> Cases_on `h` >>
  simp [raw_produce_def]
QED

Theorem raw_consume_success:
  raw_consume c traces = SOME heads ==> raw_shape traces
Proof
  simp [raw_consume_def] >> Cases_on `raw_shape traces` >> simp []
QED

Theorem raw_pipeline_join:
  bases = [1;2;3;41;61;10004;262144;265800] /\
  c = 61 /\ n = 128 /\ raw_produce n sources = traces /\
  raw_consume c traces = result ==>
  raw_pipeline bases c n sources = result
Proof
  simp [raw_pipeline_def]
QED

val mul_pre_def = cv_trans_pre "raw_mul_pre" raw_mul_def;
Theorem raw_mul_pre_ok[cv_pre]:
  !u v. raw_mul_pre u v
Proof
  PairCases_on `u` >> PairCases_on `v` >>
  simp [mul_pre_def,BlockRoundTheory.round_scaled_pre_ok] >> intLib.ARITH_TAC
QED
